\section{Proving that a commitment hides a binary string}
\label{sec:binary-proof}

The opening proof shows knowledge of \emph{some} relaxed opening and says
nothing about the message. This section adds a proof $\Pibin$ that a
commitment $C=\Com(m;r)=(t_1,t_2)$ of the base scheme hides
$m\in\{0,1\}^N$, i.e.\ a 256-bit string. $C$ is unchanged, so the
opening, equality and disclosure proofs still apply to it.

\subsection{Setting and linearization}

Notation is that of Section~\ref{sec:sec-notation}. Write
$\langle a,b\rangle=\sum_j a_jb_j$ for coefficient vectors, $\ct(a)=a_0$,
$a\circ b$ for the coefficient-wise product, $J=\sum_{j<N}X^j$ and
$\sigma\in\mathrm{Aut}(\Rq)$ for $X\mapsto X^{-1}$. Because $q\equiv5\pmod8$,
$\Rq\cong\mathbb F_{q^{N/2}}\times\mathbb F_{q^{N/2}}$; there are only two
CRT slots, so the ring-level relation $m(m-1)=0$ of
ALS20~\cite{ref:als20} would constrain only two bits. Following the
constant-coefficient technique of LNP22~\cite{ref:lnp22}, we instead linearize the
coefficient-wise relation $m\circ(m-J)=0$.

\begin{lemma}[Linearization]
\label{lem:bin-lin}
For all $a,b\in\Rq$: (i) $\ct(\sigma(a)b)=\langle a,b\rangle$; (ii) $\sigma$ is
an involution and $\langle\sigma(a),b\rangle=\langle a,\sigma(b)\rangle$;
(iii) for $\gamma,\rho\in\mathbb Z_q^N$, $u\in\Rq$ and
$R=\sigma(\gamma\circ\sigma(\rho))$,
\[
\ct\bigl(\sigma(\rho)u-Rm\bigr)=\langle\rho,\,u-\sigma(\gamma\circ m)\rangle,
\qquad
\ct\bigl(\sigma(\gamma\circ m)(m-J)\bigr)=\langle\gamma,\,m\circ(m-J)\rangle .
\]
(iv) Any quadratic form $\mathbb Z_q^N\to\mathbb Z_q$ of the shape
$m\mapsto\ct(\sum_\ell a_\ell\,\tau_\ell(m)\,b_\ell\,\tau'_\ell(m))$ with
$a_\ell,b_\ell\in\Rq$ and $\tau_\ell,\tau'_\ell\in\{\mathrm{id},\sigma\}$ is
$m^{\mathsf T}T_wm$ for a negacyclic Toeplitz $T_w$, whose diagonal is
$w_0I$.
\end{lemma}
\begin{proof}
(i) The constant coefficient of $\sigma(a)b$ collects $a_0b_0$ and, for
$i\ge1$, $(-a_i)X^{N-i}\cdot b_iX^i=a_ib_i$. (ii) $\sigma$ acts on coefficient
vectors as a symmetric signed permutation. (iii) follows from (i) and (ii).
(iv) Multiplication by $a$ is $T_a$, $\sigma$ is the signed permutation $S$
with $ST_aS=T_{\sigma(a)}$, and $T_a^{\mathsf T}=T_{\sigma(a)}$; products and sums
of negacyclic Toeplitz matrices are negacyclic Toeplitz.
\end{proof}

Item (iv) is why the prover must commit to the \emph{lifts}
$u_i=\sigma(\gamma_i\circ m)$ once $\gamma_i$ is known: a random diagonal form
$\sum_j\gamma_jm_j^2$ is not reachable from $m$ alone.

\subsection{The protocol}

Let $k$ (lift repetitions) and $k'$ (check repetitions) be parameters
($k=k'=5$). An auxiliary BDLOP key $\mathsf{ck}'=(B_1',(b'_j)_j)$ with
$B_1'=[I_n\,|\,\cdot]\in\Rq^{n\times k_{\mathrm{aux}}}$,
$b'_j=[0^n\,|\,e_j\,|\,\cdot]$, $k_{\mathrm{aux}}=n+(k+k'+1)+\lambda$, is
expanded from the parameter identifier. In each proof the prover samples a
\emph{fresh} $r'\in S_1^{k_{\mathrm{aux}}}$. The interactive protocol is:
\begin{enumerate}
\item $V$: $\gamma_1,\dots,\gamma_k\leftarrow\mathbb Z_q^N$.
\item $P$: $u_i=\sigma(\gamma_i\circ m)$, $g_t\leftarrow\{g\in\Rq:\ct(g)=0\}$;
  send $D=(B_1'r',\ \langle b'_{u_i},r'\rangle+u_i,\ \langle b'_{g_t},r'\rangle+g_t)$.
\item $V$: $\rho_{t,i}\leftarrow\mathbb Z_q^N$, $\mu_{t,i}\leftarrow\mathbb Z_q$.
\item $P$: send $h_t=g_t+E_t$ for
  $E_t=\sum_i\sigma(\rho_{t,i})u_i-R_{t,i}m+\mu_{t,i}u_i(m-J)$,
  $R_{t,i}=\sigma(\gamma_i\circ\sigma(\rho_{t,i}))$. $V$ requires $\ct(h_t)=0$.
\item $V$: $\alpha_t\leftarrow\Rq$. Both sides form
  $P(s)=\sum_t\alpha_t(g_t+E_t-h_t)=\psi m+L(s)-H$, with
  $\psi=\sum_i\beta_iu_i$, $\beta_i=\sum_t\alpha_t\mu_{t,i}$, $L$ linear and
  $H=\sum_t\alpha_th_t$.
\item $P$: masks $y\leftarrow Y$, $y'\leftarrow Y'$; with $a_j$ the projections
  $\langle b_j,y\rangle$ resp.\ $\langle b'_j,y'\rangle$ and
  $\Gamma_1=-(\psi a_m+a_\psi m+L(a))$, commit $t_G=\langle b'_G,r'\rangle+\Gamma_1$
  and announce $(t_G,w=B_1y,w'=B_1'y',v=a_\psi a_m+a_G)$.
\item $V$: $c\leftarrow\Cset$.\quad
  $P$: $z=y+cr$, $z'=y'+cr'$; abort unless both lie in $[-B,B]$.
\item $V$ accepts iff $B_1z=w+ct_1$, $B_1'z'=w'+ct_1'$ and, with masked
  openings $f_j=ct_j-\langle b_j,z\rangle$,
  \[
  v=f_\psi f_m+c\,\bigl(L(f)-cH\bigr)-f_G .
  \]
\end{enumerate}
Honestly $f_j=cs_j-a_j$, so the right-hand side equals
$c^2P(s)+c(\Gamma_1-\Gamma_1)+a_\psi a_m+a_G=v$. The non-interactive version
derives the four challenges from chained, domain-separated SHAKE-256 seeds
over $(\pid,x,C)$, $D$, $(h_t)_t$ and $(t_G,w,w',v)$ in this order; on an abort
only steps 6--7 are repeated. The proof is
$(D,(h_t)_t,t_G,\mathit{seed},z,z')$; for \texttt{bdlop-binary-128-v2} it is
43\,022 bytes.

\subsection{Security}

\begin{theorem}[Completeness]
\label{thm:bin-complete}
If $C=\Com(m;r)$ with $m\in\{0,1\}^N$ and $r\in S_1^k$, an attempt is accepted
with probability exactly
$p_{\mathrm{acc}}=((2B+1)/2\gamma)^{(k+k_{\mathrm{aux}})N}$
($0.0112$ for \texttt{bdlop-binary-128-v2}), every accepted transcript
verifies, and the prover fails to output a proof within $\Amax=16\,384$
attempts with probability $(1-p_{\mathrm{acc}})^{\Amax}<2^{-266}$.
\end{theorem}
\begin{proof}
$\ct(E_t)=0$ by Lemma~\ref{lem:bin-lin}(iii) with $u_i=\sigma(\gamma_i\circ m)$
and $m\circ(m-J)=0$, so $\ct(h_t)=0$. $P(s)=0$ by construction, and the
verification identity is the computation above. Acceptance is the
uniform-masking argument of Lemma~\ref{lem:hvzk} applied to the
$(k+k_{\mathrm{aux}})N$ coefficients of $(z,z')$, using
$\norm{cr}_\infty,\norm{cr'}_\infty\le\kappa$.
\end{proof}

\begin{lemma}[Consistent masked openings]
\label{lem:bin-consistent}
Let three accepting transcripts share the messages of steps 1--6 and have
pairwise distinct challenges $c,c',c''$. Then either one obtains a nonzero
$x$ with $B_1x=0$ or $B_1'x=0$ and $\norm{x}_\infty\le8\kappa B$, or there are
$\bar f\in\Cdiff$, relaxed openings $(\bar r,\bar f)$ of $C$ and
$(\bar r',\bar f)$ of $D$ with messages $\bar m,\bar s$, and vectors
$y^\star,y'^\star$ such that $z=y^\star+c\bar f^{-1}\bar r$ (likewise for
$z'$) for all three transcripts. Consequently
$v=c^2P(\bar s)+cX+Y$ for fixed $X,Y\in\Rq$ and all three challenges, and
$P(\bar s)=0$.
\end{lemma}
\begin{proof}[Proof sketch]
The first two transcripts give relaxed openings with $\bar f=c-c'$ and
$\bar r=z-z'$, $\bar r'=z'_{(1)}-z'_{(2)}$, as in Lemma~\ref{lem:fork} and the
equality proof (Lemma~\ref{lem:eq-ss}). Put $y^\star_c=z_c-c\bar f^{-1}\bar r$;
then $B_1y^\star_c=w$ for every accepting $c$, and
$\bar f(y^\star_c-y^\star_{c''})=\bar f(z_c-z_{c''})-(c-c'')\bar r$ is a vector
of norm at most $2\kappa\cdot2B+2\kappa\cdot2B=8\kappa B$ in the kernel of
$B_1$; it is zero unless it is an \MSIS{} solution, and then $y^\star$ is
independent of $c$ because $\bar f$ is invertible
(Lemma~\ref{lem:invertible}). The same holds for $B_1'$. Substituting into
$f_j$ gives $f_j=c\bar s_j-\bar a_j$ with $\bar a_j$ independent of $c$, so the
verification equation is a polynomial identity of degree two in $c$ with
leading coefficient $P(\bar s)$. Subtracting it at $c,c'$ and $c',c''$ and
using the invertibility of $c-c'$, $c'-c''$ and $c-c''$ gives $P(\bar s)=0$.
\end{proof}

\begin{theorem}[Round-by-round soundness]
\label{thm:bin-rbr}
Let $C$ be a commitment whose relaxed-opening message $\bar m$ (unique by
Theorem~\ref{thm:binding}) is not binary. For any prover of the interactive
protocol, the probability that the verifier accepts and no \MSIS{} solution
for $B_1$ or $B_1'$ of norm $\le8\kappa B$ is found is at most
\[
\erbr=q^{-k}+q^{-k'}+q^{-N/2}+2/|\Cset| ,
\]
and each of the four challenge rounds individually contributes at most the
corresponding term. For \texttt{bdlop-binary-128-v2},
$\erbr\le2^{-149.99}$.
\end{theorem}
\begin{proof}[Proof sketch]
Fix the extracted messages; they are determined by $C$ and $D$ up to
\MSIS{} (binding with respect to relaxed openings, for both keys).
\emph{Lift round.} Let $\Delta_i=\bar u_i-\sigma(\gamma_i\circ\bar m)$,
$Q_i=\ct(\bar u_i(\bar m-J))$ and $e=(\Delta_i,Q_i)_i$. If $e=0$ then
$Q_i=\langle\gamma_i,\bar m\circ(\bar m-J)\rangle=0$ for all $i$ by
Lemma~\ref{lem:bin-lin}(iii); since $\bar m$ is fixed before $\gamma$ and
$\bar m\circ(\bar m-J)\neq0$, this has probability $q^{-k}$, whatever $D$ the
prover sends afterwards. \emph{Check round.} By Lemma~\ref{lem:bin-lin},
$\ct(\bar g_t+E_t(\bar s))=\delta_t+\varphi_t(e)$ with
$\delta_t=\ct(\bar g_t)$ and
$\varphi_t(e)=\sum_i\langle\rho_{t,i},\Delta_i\rangle+\mu_{t,i}Q_i$. Both $e\neq0$
and $\delta_t$ are fixed by $D$ before $(\rho,\mu)$ is drawn, and
$\varphi_t(e)$ is uniform on $\mathbb Z_q$ for each $t$, independently; hence
$\delta_t+\varphi_t(e)=0$ for all $t$ with probability $q^{-k'}$. Otherwise some
$t$ has $\ct(\bar g_t+E_t)\ne0=\ct(h_t)$, i.e.\ $\bar P_t=\bar g_t+E_t-h_t\neq0$.
\emph{Combine round.} The $\bar P_t$ are fixed before $\alpha$. In a CRT slot
$\mathbb F_{q^{N/2}}$ where some $\bar P_{t_0}\neq0$, $\sum_t\alpha_t\bar P_t$ is
uniform, so it vanishes with probability $q^{-N/2}$. \emph{Challenge round.}
If $P(\bar s)\ne0$, pick a slot $\phi$ where it is nonzero. The polynomial
$c^2P(\bar s)+cX+Y-v$ is nonzero of degree two modulo $\phi$, so it has at
most two roots in the field $\Rq/\phi$; as $c-c'$ is invertible for distinct
challenges, $c\mapsto c\bmod\phi$ is injective on $\Cset$, and at most two
challenges are roots. By Lemma~\ref{lem:bin-consistent} an accepting
transcript needs a root.
\end{proof}

\paragraph{The non-interactive proof.}
Write $H_1,\dots,H_4$ for the four domain-separated oracles (lift, check,
combine, challenge), modelled as independent random oracles, and
$\varepsilon_1=q^{-k}$, $\varepsilon_2=q^{-k'}$, $\varepsilon_3=q^{-N/2}$,
$\varepsilon_4=2/|\Cset|$ for the four terms of Theorem~\ref{thm:bin-rbr}
(we model $c=\mathsf{SampleInBall}(H_4(\cdot))$ as uniform on $\Cset$). The
seeds are chained: the input of $H_{i+1}$ contains the output $s_i$ of $H_i$.
For a relaxed opening with messages $\bar s$ of the rows hashed into an
$H_i$-query, call the answer of that query \emph{bad} if it is the bad event
of round $i$ in the proof of Theorem~\ref{thm:bin-rbr}: all
$\langle\gamma_i,\bar v\rangle=0$ while $\bar v=\bar m\circ(\bar m-J)\ne0$
($i=1$); $\delta_t+\varphi_t(e)=0$ for all $t$ while $e\ne0$ ($i=2$);
$\sum_t\alpha_t\bar P_t=0$ while some $\bar P_t\ne0$ ($i=3$). Each of these
depends only on the query input (it contains $C$, the rows of $D$ sent so
far, $(h_t)_t$ and, through the chained seed, the earlier challenges) and on
the answer, and for every fixed $\bar s$ it has probability at most
$\varepsilon_i$ over the answer.

\begin{theorem}[Knowledge soundness of the non-interactive proof]
\label{thm:bin-ks}
Let $\mathcal A$ make at most $Q$ queries to $H_1,\dots,H_4$ in total and
output $(x,C,\pi)$ that the verifier accepts with probability $\varepsilon$.
There is an extractor $\mathcal E$ that runs $\mathcal A$ an expected number of
at most $3+2Q$ times, and for $i=1,2,3$ an algorithm $\mathcal F_i$ that runs
$\mathcal E$ twice, such that $\mathcal E$ outputs a relaxed opening of $C$
whose message is binary with probability at least
\[
\varepsilon-(Q+1)\sum_{i=1}^{4}\varepsilon_i-\sum_{i=1}^{3}\sqrt{(Q+1)\,\alpha_i}
-\beta-\frac{Q^2}{2^{256}},
\]
where $\beta$ (resp.\ $\alpha_i$) is the probability that $\mathcal E$
(resp.\ $\mathcal F_i$) outputs an \MSIS{} solution of $\ell_\infty$ norm at
most $8\kappa B$ for $B_1$ or $B_1'$. In particular, if $C$ has no relaxed
opening with a binary message, then
$\varepsilon\le(Q+1)\erbr+\sum_i\sqrt{(Q+1)\alpha_i}+\beta+Q^22^{-256}$ with
$\erbr\le2^{-149.99}$ for \texttt{bdlop-binary-128-v2}.
\end{theorem}
\begin{proof}[Proof sketch]
\emph{Aborts.} The prover repeats only steps~6--7 after an abort; the seeds
$s_1,s_2,s_3$, the commitment $D$ and the masks $(h_t)_t$ are fixed before
the first attempt. To the verifier and to a cheating prover, a rejected
attempt is an $H_4$-query whose answer is discarded, so it is counted in $Q$
and needs no further treatment for soundness. (For zero knowledge, rounds
1--3 never abort, so the simulator obtains $\gamma,\rho,\mu,\alpha$ by plain
oracle queries on its simulated $D$ and $h$, and only the aborting last round
is simulated by programming $H_4$, exactly as for the opening proof in
Lemma~\ref{lem:sim}, following~\cite{ref:barbosa,ref:devevey}.)

\emph{Query order.} Without loss of generality $\mathcal A$ queries the four
oracle inputs of its output (one more query each). Because the input of
$H_{i+1}$ contains the 256-bit output of $H_i$, except with probability
$Q^2/2^{256}$ these four queries occur in the order $H_1,H_2,H_3,H_4$ and the
$H_4$-query is the last of them.

\emph{The last round.} Fix $\mathcal A$'s coins and the answers to all queries
before the critical $H_4$-query. Its input fixes the prefix
$\mathit{pre}=(x,C,D,(h_t)_t,s_3)$ and the announcement $(t_G,w,w',v)$, and
the remaining protocol is a $\Sigma$-protocol for the statement $\mathit{pre}$
with challenge $c$ and response $(z,z')$. By Lemma~\ref{lem:bin-consistent} it
is $3$-special sound for the relation ``relaxed openings of $C$ and $D$ with
a common $\bar f\in\Cdiff$ whose messages satisfy $P_{\mathit{pre}}(\bar s)=0$,
or an \MSIS{} solution of norm $\le8\kappa B$''. The Fiat--Shamir extractor
of Attema, Fehr and Klo{\ss}~\cite{ref:afk22} for $k$-special-sound
$\Sigma$-protocols ($k=3$; it rewinds only the critical query of the
challenge oracle and treats the statement and every other oracle as part of
$\mathcal A$) runs $\mathcal A$ an expected $k+Q(k-1)=3+2Q$ times and
succeeds with probability at least
$(\varepsilon-(Q+1)\varepsilon_4)/(1-\varepsilon_4)\ge\varepsilon-(Q+1)\varepsilon_4$.

\emph{The statistical rounds.} If $\mathcal E$ outputs openings with messages
$\bar s$ and none of the three round challenges of the output is bad for
$\bar s$, the chain of Theorem~\ref{thm:bin-rbr} applies verbatim: a
non-binary $\bar m$ would give $P_{\mathit{pre}}(\bar s)\ne0$. It remains to
bound the event $\mathsf{Bad}_i$ that the round-$i$ challenge is bad for the
extracted $\bar s$. The difficulty is that $\bar s$ is extracted \emph{after}
the challenge and is fixed by the earlier rows only computationally (binding
w.r.t.\ relaxed openings). Fix $j\le Q$ and everything before the $j$-th
$H_i$-query (the ``prefix''); let $\xi$ be the rest of the randomness of
$\mathcal E$, including the answer $\gamma^{(j)}$ to that query. For each
message vector $m$ of the rows hashed into the query let
$p_m=\Pr_\xi[\mathcal E \text{ outputs openings, the output uses query }j,
\bar s|_{\mathrm{rows}}=m]$, $P=\sum_mp_m$, and let $m^\star$ maximise
$p_m$; $m^\star$ depends on the prefix only, hence is independent of
$\gamma^{(j)}$. Then
\[
\Pr_\xi[\mathsf{Bad}_i\wedge\text{query }j]
\le\Pr[\gamma^{(j)}\text{ bad for }m^\star]+\Pr_\xi[\bar s|_{\mathrm{rows}}\ne m^\star]
\le\varepsilon_i+(P-p_{m^\star}).
\]
$\mathcal F_i$ runs $\mathcal E$, and if the output uses the $j$-th
$H_i$-query it reruns $\mathcal E$ from that query with fresh $\xi'$; it
succeeds if both runs use query $j$ and their messages on the hashed rows
differ. Two relaxed openings $(\bar r,\bar f)$, $(\bar r',\bar f')$ of the
same $t_1$ (or $t_1'$) with different messages give the \MSIS{} solution
$\bar f'\bar r-\bar f\bar r'\ne0$ of norm $\le8\kappa B$, as in
Theorem~\ref{thm:binding}. Given the prefix, the success probability is
$P^2-\sum_mp_m^2\ge P(P-p_{m^\star})\ge(P-p_{m^\star})^2$. Summing over $j$,
with Jensen's and the Cauchy--Schwarz inequality,
$\sum_j\mathbb E[P-p_{m^\star}]\le\sqrt{(Q+1)\alpha_i}$, so
$\Pr[\mathsf{Bad}_i]\le(Q+1)\varepsilon_i+\sqrt{(Q+1)\alpha_i}$.
Collecting the terms gives the bound.
\end{proof}

\noindent\emph{Status and remaining gaps.} This closes the composition
that earlier drafts left open: aborts touch only the last round, the
statistical rounds lose the usual factor $Q+1$ of
round-by-round soundness~\cite{ref:ccHLRR19}, and only the last oracle is
forked. What remains is (i) the \MSIS{} terms $\sqrt{(Q+1)\alpha_i}$, which
carry the same square-root (forking) loss as the opening proof
(Lemma~\ref{lem:fork}) and make the reduction non-tight;
(ii) the use of the Attema--Fehr--Klo{\ss} $\Sigma$-protocol bound with a
statement (the prefix) chosen adaptively by $\mathcal A$ through other oracles,
which we believe is covered by their adaptive formulation but have not
checked line by line; (iii) the modelling of
$\mathsf{SampleInBall}\circ H_4$ as a uniform challenge (the seed has 256
bits and $|\Cset|=2^{257}$); and (iv) a classical-ROM argument only, as for
every theorem here. The proof is a sketch and has not been checked
independently.

\begin{theorem}[Honest-verifier zero knowledge]
\label{thm:bin-zk}
There is a simulator that, given $(x,C)$ only and programming the random
oracle at four points, outputs proofs whose distribution differs from honest
proofs by at most $\Adv^{\MLWE}_{\mathsf{aux}}$ (hiding of the auxiliary
instance: secret rank $\lambda$, $n+k+k'+1$ ring samples, ternary secret and
error) plus the programming-collision terms of Lemma~\ref{lem:sim}, for any
number of proofs on the same $C$.
\end{theorem}
\begin{proof}[Proof sketch]
(1) Accepted $(z,z')$ are uniform on $[-B,B]^{(k+k_{\mathrm{aux}})N}$ and
independent of $(r,r')$ (Lemma~\ref{lem:hvzk}); sample them directly and
program the oracles, the announcement having min-entropy through $w$ and $w'$.
(2) $t_G$'s message $\Gamma_1$ is a function of $z'$, $c$, the public rows of
$D$ and the witness (since $a_j=\langle b'_j,z'\rangle-c(t'_j-s_j)$), so a
reduction that knows $m$ but not $r'$ can produce $D$ and $t_G$ from an
\MLWE{} challenge for $\mathsf{ck}'$; replace them by uniform values.
(3) Then $g_t$ occurs only in $h_t$, which is uniform on $\{\ct=0\}$; the
simulator samples $h_t$ there. The verification equations determine
$w,w',v$. The base randomness $r$ is used across proofs only through
rejection-sampled responses, and $r'$ is fresh in each proof, so the
hybrids compose over many proofs.
\end{proof}

\subsection{Binding the proof to the payload encoding}
\label{sec:payload-range}

$\Pibin$ applies to a commitment of a 256-bit string, whereas the payload
commitments of Section~\ref{sec:sec-notation} encode $\mathrm{len}(p)\,\|\,p$
with three bytes per coefficient. A variant $\Pi_{\mathsf{pay}}$
(\texttt{payload\_range\_proof.py}, set \texttt{bdlop-payload-range-128-v1})
proves a statement about the payload commitment itself.

\paragraph{Statement.} Public input $(x,L,C)$ with $C$ an unchanged base
commitment and a public length $L\le128$ bytes. Write
$\mathrm{Enc}(p)=e_L+A\beta(p)$, where $\beta(p)\in\{0,1\}^{\ell N}$ are the
bits of $p$ (most significant first, $\ell=\lceil8L/N\rceil$ binary
polynomials $\beta_j$), $e_L=\mathrm{Enc}(0^L)$ carries the length field, and
$A$ is the public $\mathbb Z_q$-linear map that sends payload bit $s$ to
coefficient $\lfloor(2+\lfloor s/8\rfloor)/3\rfloor$ with weight
$2^{8(2-(2+\lfloor s/8\rfloor)\bmod3)+7-s\bmod8}$. $\Pi_{\mathsf{pay}}$ proves
knowledge of a relaxed opening of $C$ whose message is $\mathrm{Enc}(p)$ for
some $p\in\{0,1\}^{8L}$: every coefficient lies in $[0,2^{24})$ and is the
packing of payload bytes, coefficient~$0$ carries $L$, and all coefficients
beyond $\lceil(L+2)/3\rceil$ vanish. It reveals $L$ and nothing else about $p$.

\paragraph{Protocol.} It is $\Pibin$ with three changes. (1) The binary
polynomials live in the first $\ell$ slots of the auxiliary commitment $D$,
whose rows $D_0=(t_1',t_{\beta_j})$ are hashed into $H_1$ together with
$(\pid,x,L,C)$; the lifts and masks are added afterwards as further rows of
$D$ under the same randomness $r'$. (2) There are $k$ lifts per binary
polynomial, $u_{j,i}=\sigma(\gamma_{j,i}\circ\beta_j)$. (3) Each check
functional gets a link term with a fresh $\rho^L_t\in\mathbb Z_q^N$:
$E_t$ gains $\sigma(\rho^L_t)m-\sum_j\sigma(A_j^{\mathsf T}\rho^L_t)\beta_j
-\langle\rho^L_t,e_L\rangle$, whose constant coefficient is
$\langle\rho^L_t,\,m-A\beta-e_L\rangle$ by Lemma~\ref{lem:bin-lin}(i). The
combined relation is $\sum_j\psi_j\beta_j+L(s)-H'=0$ with
$\psi_j=\sum_i\beta_{j,i}u_{j,i}$, proved with one garbage polynomial as before.

\begin{theorem}[Payload range proof]
\label{thm:pay}
Completeness holds as in Theorem~\ref{thm:bin-complete} with
$k_{\mathrm{aux}}=n+\ell(k+1)+k'+1+\lambda$ (acceptance $0.0100$ for
$\ell=1$, $0.0012$ for $\ell=4$; cap $2^{17}$ attempts). Theorems
\ref{thm:bin-rbr} and \ref{thm:bin-ks} hold for $\Pi_{\mathsf{pay}}$ with the
same four round errors and with ``binary message'' replaced by ``message
$\mathrm{Enc}(p)$ for some $p\in\{0,1\}^{8L}$''. Zero knowledge holds as in
Theorem~\ref{thm:bin-zk}; the simulator is given $(x,L,C)$.
\end{theorem}
\begin{proof}[Proof sketch]
Lift round: if some $\bar\beta_j$ is not binary, the $k$ functionals
$\langle\gamma_{j,i},\bar\beta_j\circ(\bar\beta_j-J)\rangle$ all vanish with
probability $q^{-k}$, as $\bar\beta$ is fixed by $D_0$ before $\gamma$. Check
round: the error vector $e$ now also contains
$e_{\mathrm{link}}=\bar m-A\bar\beta-e_L$, which is fixed by $C$ and $D_0$
before $\rho^L$, and $\varphi_t(e)$ gains
$\langle\rho^L_t,e_{\mathrm{link}}\rangle$; it stays uniform when $e\ne0$.
The combine and challenge rounds and the extraction are unchanged
(Lemma~\ref{lem:bin-consistent} for $C$ and $D$). If no bad event occurs,
$\bar\beta$ is binary and $\bar m=A\bar\beta+e_L$ holds in $\mathbb Z_q^N$;
since $A\bar\beta+e_L$ has integer coefficients below $2^{24}<q$, $\bar m$ is
exactly $\mathrm{Enc}(p)$ for the $p$ whose bits are the first $8L$ entries
of $\bar\beta$.
\end{proof}

For \texttt{bdlop-payload-range-128-v1} the proof has $44\,558$ bytes for
$L\le32$ ($\ell=1$) and grows by $9\,216$ bytes per further 32 payload bytes
(72\,206 bytes at $L=128$). The auxiliary \MSIS{} ($n\times25$ to
$n\times43$) and \MLWE{} (18 to 36 samples) instances give the same pinned
estimates as the base set
(\texttt{research/bdlop\_params/bdlop\_payload\_range\_128\_l\{1,4\}.estimate.json}).
Two alternatives are worse. Encoding the payload in binary from the start
(one bit per coefficient) lets $\Pibin$ apply directly, at
$6\,720+43\,022$ bytes per 32 payload bytes, but needs a new envelope format
and disclosure encoding. A range proof of all $256$ payload coefficients
($\ell=24$ bit-planes) would need $177$ response polynomials,
$256\,526$ bytes and an acceptance probability of $8\cdot10^{-10}$ per attempt
with the fixed mask bound $\gamma=2^{17}$, so it is impractical without a
larger auxiliary mask bound and a new \MSIS{} estimate.

\begin{remark}[Limits]
The theorems are classical-ROM statements; no QROM claim is made. The
auxiliary key's \MSIS{} ($n\times24$ instead of $n\times14$) and \MLWE{}
($17$ instead of $7$ samples) instances give the same levels as the base set
under the pinned estimator (\MSIS{} $\beta=480$, 140/127 bits; primal \MLWE{}
$\beta=580$, 169/153 bits; dual $\beta=578$), since the optimal \MSIS{}
sub-dimension ($\approx3\,022$) is below both widths
(\texttt{research/bdlop\_params/bdlop\_binary\_128\_v2.estimate.json}); the
dual-hybrid caveat of Section~\ref{sec:proof} applies equally. The
soundness error $2^{-150}$ per round is below the $2^{-256}$ of the opening
proof; $k=k'=9$ would match it at about 12\,KB more. $\Pi_{\mathsf{pay}}$
makes the length $L$ public and covers payloads of at most 128 bytes; a
hidden length would need a proof that the bytes after an unknown length are
zero, and statements about the payload \emph{value} (beyond its encoding)
are not proved, although any public affine constraint on the payload bits
could be added as one more link functional.
\end{remark}
